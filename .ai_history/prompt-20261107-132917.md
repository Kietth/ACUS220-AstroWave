// Composición del vuelo de intro. Módulo puro (sin three.js): tiempos, cámara y colocación de cuerpos.
//
// Idea: cada cuerpo se define por el INSTANTE en que es "protagonista" (t) y por DÓNDE debe estar
// en pantalla en ese instante (nx, ny en coordenadas normalizadas -1..1). A partir de ahí se calcula
// su posición en el mundo según el aspecto de la pantalla, de modo que la composición se mantiene
// en cualquier formato (móvil vertical, 16:9, ultra-ancho).

export const DEG = Math.PI / 180;

// ── Tiempos (se conservan: 6,4 s en total) ──────────────────────────────────────────────────────
export const VMAX = 800;
export const T = { accel: 1.4, cruise: 2.4, decel: 2.6 };
export const T_SETTLE = T.accel + T.cruise;
export const T_END = T_SETTLE + T.decel;

export const speedAt = (t) => {
  if (t < T.accel) return VMAX * (t / T.accel) ** 2;
  if (t < T_SETTLE) return VMAX;
  return VMAX * (1 - (t - T_SETTLE) / T.decel) ** 2;
};

// Posición z de la cámara (integral exacta de speedAt).
export const camZAt = (t) => {
  let d = (VMAX * Math.min(t, T.accel) ** 3) / (3 * T.accel ** 2);
  if (t > T.accel) d += VMAX * Math.min(t - T.accel, T.cruise);
  if (t > T_SETTLE) {
    const u = Math.min(t - T_SETTLE, T.decel) / T.decel;
    d += ((VMAX * T.decel) / 3) * (1 - (1 - u) ** 3);
  }
  return -d;
};

// ── Cámara ──────────────────────────────────────────────────────────────────────────────────────
// FOV vertical de vuelo. 46° (antes 70° + 9° de "kick"): con 70° las esferas se deformaban en
// elipses en los bordes (se veía en el sol enorme del vídeo). Por debajo de ~55° la distorsión es mínima.
export const FLIGHT_FOV = 46;
export const FOV_KICK = 4;

// Distancia (en unidades del mundo) a la que un cuerpo está exactamente en su punto (nx, ny).
export const D_REF = 520;

// Límites de formato para los que se calcula la composición: de móvil muy estrecho a ultra-ancho 2,2:1.
// Fuera de ese rango se usa el límite (los cuerpos no se alejan más hacia los extremos).
const ASPECT_MIN = 0.45;
const ASPECT_MAX = 2.2;

export function frustum(aspect, fov = FLIGHT_FOV) {
  const a = Math.min(Math.max(aspect, ASPECT_MIN), ASPECT_MAX);
  const half = Math.tan((fov * DEG) / 2);
  // La cámara trata `fov` como horizontal cuando la pantalla es vertical (ver World.updateCamera).
  return a >= 1 ? { tanH: half * a, tanV: half } : { tanH: half, tanV: half / a };
}

// ── Cuerpos de paso: un zigzag Z a través de los puntos de fuerza de la regla de tercios ─────────
// nx/ny = posición objetivo en pantalla cuando el cuerpo es protagonista (los tercios están en ±0,33).
// Alternan izquierda/derecha y arriba/abajo para que el ojo "zigzaguee" sin que nada tape a nada,
// y desde lejos todos convergen hacia el punto de fuga: líneas de guía que llevan al centro.
export const BODIES = [
  { kind: 'planet', t: 0.6, nx: -0.42, ny: 0.18, r: 50, cols: ['#c9774a', '#a85a35', '#e0a070'], bands: 14, lean: 0.28, tilt: 0.22 },
  { kind: 'star', t: 1.3, nx: 0.46, ny: -0.16, r: 58, color: 0xffd27a },
  { kind: 'planet', t: 2.0, nx: -0.5, ny: -0.2, r: 44, cols: ['#d9c08a', '#bfa06a', '#efe0b8', '#a88a55'], bands: 22, ring: true, lean: 0.55, tilt: 0.3 },
  { kind: 'star', t: 2.6, nx: 0.44, ny: 0.22, r: 54, color: 0xa8c8ff },
  { kind: 'star', t: 3.1, nx: -0.4, ny: 0.02, r: 62, color: 0xff5a2a },
];

// Aparición por distancia: los cuerpos lejanos no existen hasta que están "cerca" (como la bruma
// atmosférica). Sin esto, por perspectiva, todos se amontonan en el punto de fuga al inicio del vuelo.
export const FADE_FAR = 2300;
export const FADE_NEAR = 1500;
export const depthFade = (dist) => {
  const k = Math.min(1, Math.max(0, (FADE_FAR - dist) / (FADE_FAR - FADE_NEAR)));
  return k * k * (3 - 2 * k);
};

// Salida de cuadro: un cuerpo que ya está saliendo por el lateral se disuelve (|ndcX| de 1,0 a 1,35).
// Evita que una esfera enorme se deforme y se corte contra el borde (el "sol" del vídeo original).
export const edgeFade = (ndcX) => {
  const k = Math.min(1, Math.max(0, (Math.abs(ndcX) - 1.0) / 0.35));
  return 1 - k * k * (3 - 2 * k);
};

// Holgura mínima respecto al eje de vuelo, relativa al radio: limita cuánto se "abalanza" un cuerpo
// al pasar (con r/x = 0,28 nunca ocupa más de ~1/4 de la altura al salir de cuadro).
const clearanceOf = (d) => (d.ring ? 5 : 3.6) * d.r;

export function placeBodies(aspect) {
  const { tanH, tanV } = frustum(aspect);
  return BODIES.map((d) => {
    const side = d.nx < 0 ? -1 : 1;
    return {
      x: side * Math.max(Math.abs(d.nx) * tanH * D_REF, clearanceOf(d)),
      y: d.ny * tanV * D_REF,
      z: camZAt(d.t) - D_REF,
    };
  });
}

// ── Destino: el horizonte de un planeta que sube desde abajo durante la frenada ──────────────────
// Va pegado a la cámara (no al mundo): así su posición final no depende de dónde termine el vuelo,
// tampoco si el usuario salta la intro.
export const DEST = {
  r: 2400, // radio: tan grande que se lee como horizonte, no como una bola
  c: 2800, // distancia cámara → centro
  top: -0.4, // borde superior del casquete en pantalla (NDC). 30 % de la altura queda ocupada.
  color: 0x1a8f88,
};
// Fracción de la altura de pantalla que ocupa el horizonte (la usa el CSS: --horizon).
export const HORIZON_FRACTION = (1 + DEST.top) / 2;

// Elevación (rad, negativa = bajo el eje) del borde superior de la esfera cuyo centro está en (y, z).
const limbElevation = (y, z) => -Math.atan2(-y, -z) + Math.asin(DEST.r / Math.hypot(y, z));

export function placeDest(aspect) {
  const { tanV } = frustum(aspect);
  const elevation = Math.atan(DEST.top * tanV); // dirección (negativa) del borde superior
  const beta = Math.asin(DEST.r / DEST.c); // radio angular de la esfera
  const alpha = beta - elevation; // ángulo bajo el eje al que está el centro
  const y = -DEST.c * Math.sin(alpha);
  const z = -DEST.c * Math.cos(alpha);

  // Cuánto hay que bajarlo para que, al empezar, el borde quede fuera de cuadro (con margen para el
  // "kick" de FOV). Se calcula por formato: en vertical el horizonte está mucho más cerca de la cámara.
  const hidden = -Math.atan(tanV * 1.45);
  let lo = 0;
  let hi = 20000;
  for (let i = 0; i < 40; i += 1) {
    const mid = (lo + hi) / 2;
    if (limbElevation(y - mid, z) > hidden) lo = mid;
    else hi = mid;
  }
  return { y, z, rise: hi, exit: hi * 0.8 };
}

import { describe, expect, it } from 'vitest';
import { BODIES, DEST, T, T_SETTLE, camZAt, depthFade, edgeFade, placeBodies, placeDest, speedAt } from './flightLayout.js';

const ASPECTS = [390 / 844, 1, 16 / 9, 21 / 9];

describe('flightLayout', () => {
  it('camZAt es la integral de speedAt', () => {
    const marks = [1, 2, 3.8, 5, 6.4];
    const dt = 0.001;
    let z = 0;
    let t = 0;
    marks.forEach((mark) => {
      for (; t < mark - 1e-9; t += dt) z -= speedAt(t + dt / 2) * dt;
      expect(camZAt(mark)).toBeCloseTo(z, 0);
    });
    expect(camZAt(0)).toBeCloseTo(0, 6);
  });

  it('los cuerpos alternan izquierda y derecha (zigzag)', () => {
    BODIES.slice(1).forEach((b, i) => expect(Math.sign(b.nx)).toBe(-Math.sign(BODIES[i].nx)));
  });

  it('cada cuerpo pasa con holgura lateral suficiente en cualquier formato', () => {
    ASPECTS.forEach((a) => {
      placeBodies(a).forEach((p, i) => {
        const min = (BODIES[i].ring ? 5 : 3.6) * BODIES[i].r;
        expect(Math.abs(p.x)).toBeGreaterThanOrEqual(min - 1e-6);
      });
    });
  });

  it('todos los cuerpos quedan atrás cuando empieza la llegada', () => {
    placeBodies(16 / 9).forEach((p) => expect(camZAt(T_SETTLE + 0.5)).toBeLessThan(p.z));
  });

  it('la aparición por distancia y la salida de cuadro son suaves y acotadas', () => {
    expect(depthFade(3000)).toBe(0);
    expect(depthFade(1000)).toBe(1);
    expect(edgeFade(0.5)).toBe(1);
    expect(edgeFade(2)).toBe(0);
    expect(edgeFade(1.15)).toBeGreaterThan(0);
    expect(edgeFade(1.15)).toBeLessThan(1);
  });

  it('el horizonte llega con su borde en DEST.top y parte fuera de cuadro', () => {
    const elevation = (y, z) => -Math.atan2(-y, -z) + Math.asin(DEST.r / Math.hypot(y, z));
    ASPECTS.forEach((a) => {
      const { y, z, rise } = placeDest(a);
      const tanV = Math.tan((46 * Math.PI) / 360) * (a < 1 ? 1 / Math.max(a, 0.45) : 1);
      expect(Math.tan(elevation(y, z)) / tanV).toBeCloseTo(DEST.top, 2);
      expect(Math.tan(elevation(y - rise, z)) / tanV).toBeLessThan(-1);
    });
  });

  it('la llegada dura lo que la frenada', () => {
    expect(T.decel).toBeCloseTo(2.6);
  });
});

function planetTexture(cols, bands, seed) {
  const c = document.createElement('canvas');
  c.width = 512;
  c.height = 256;
  const g = c.getContext('2d');
  let s = seed;
  const rnd = () => (s = (s * 9301 + 49297) % 233280) / 233280;
  for (let i = 0; i < bands; i++) {
    g.fillStyle = cols[Math.floor(rnd() * cols.length)];
    g.fillRect(0, (i * 256) / bands, 512, 256 / bands + 1);
  }
  for (let i = 0; i < 260; i++) {
    g.fillStyle = `rgba(${rnd() > 0.5 ? '255,255,255' : '0,0,0'},${rnd() * 0.12})`;
    g.beginPath();
    g.ellipse(rnd() * 512, rnd() * 256, rnd() * 50 + 6, rnd() * 8 + 2, 0, 0, 7);
    g.fill();
  }
  return new THREE.CanvasTexture(c);
}

      o.geometry?.dispose();
      if (o.material) {
        o.material.map?.dispose();
        o.material.dispose();
      }
    });
    this.glow.dispose();
    this.renderer.dispose();
    this.renderer.forceContextLoss();
    this.labelLayer.replaceChildren();
  }


  tween(obj, key, to, ms, { delay = 0, ease = EASE.inOut, onDone } = {}) {
    this.tweens = this.tweens.filter((t) => !(t.obj === obj && t.key === key));
    if (this.reducedMotion || ms <= 0) {
      obj[key] = to;
      onDone?.();
      return;
    }
    this.tweens.push({ obj, key, from: obj[key], to, t0: performance.now() + delay, ms, ease, onDone });
  }

  runTweens(now) {
    const done = [];
    for (const t of this.tweens) {
      if (now < t.t0) continue;
      const k = Math.min(1, (now - t.t0) / t.ms);
      t.obj[t.key] = t.from + (t.to - t.from) * t.ease(k);
      if (k >= 1) done.push(t);
    }
    if (done.length) {
      this.tweens = this.tweens.filter((t) => !done.includes(t));
      done.forEach((t) => t.onDone?.());
    }
  }


  settle() {
    if (this.fl.settled) return;
    this.fl.settled = true;
    this.onSettle?.();
  }

  finishFlight() {
    const F = this.fl;
    F.state = 'done';
    F.v = 0;
    this.arrive = 1;
    this.settle();
    this.drift = !this.reducedMotion && !this.exploring;
    this.tween(this, 'flightAlpha', 0, 500, { ease: EASE.exit });
  }

  stepFlight(dt) {
    const F = this.fl;
    if (F.state === 'running') {
      F.t += dt;
      F.v = speedAt(F.t);
      this.arrive = clamp01((F.t - T_SETTLE) / T.decel);
      if (!F.settled && F.t >= T_SETTLE) this.settle();
      if (F.t >= T_END) this.finishFlight();
    } else if (F.state === 'braking') {
      F.s += dt / BRAKE_S;
      F.v = F.v0 * (1 - Math.min(F.s, 1)) ** 2;
      this.arrive = Math.max(this.arrive, Math.min(F.s, 1));
      if (F.s >= 1) this.finishFlight();
    } else {
      return;
    }
    this.speedFrac = Math.min(1, F.v / VMAX);
    this.anchor.z -= F.v * dt;
    this.anchor.x = Math.sin(this.time * 1.2) * 1.2;
    this.anchor.y = Math.cos(this.time * 0.9) * 0.8;
  }

  // Opacidad de cada cuerpo de paso = desvanecimiento global × aparición por distancia × salida de cuadro.
  updateBodies() {
    const camZ = this.anchor.z;
    const tanH = Math.tan((this.camera.fov * DEG) / 2) * this.camera.aspect;
    let any = false;
    this.bodies.forEach((b) => {
      const dz = camZ - b.group.position.z;
      let a = 0;
      if (dz > 1) a = this.flightAlpha * depthFade(dz) * edgeFade((b.group.position.x - this.anchor.x) / (dz * tanH));
      if (Math.abs(a - b.alpha) >= 0.002 || b.alpha === -1) {
        b.alpha = a;
        b.group.visible = a > 0.003;
        for (const { mat, base, uniform } of b.fade) {
          if (uniform) mat.uniforms.uAlpha.value = a;
          else mat.opacity = base * a;
        }
      }
      any = any || b.group.visible;
    });
    return any;
  }

  applyDestAlpha(a) {
    for (const m of this.destMats) m.uniforms.uAlpha.value = a;
    this.dest.visible = a > 0.003;
    this.appliedDestAlpha = a;
  }

  buildFlight() {
    this.flight = new THREE.Group();
    this.scene.add(this.flight);
    this.spinners = [];
    this.starMats = [];
    this.glow = glowTexture();
    this.warp = new WarpField();
    this.warp.reset(0);
    this.scene.add(this.warp.object);

    this.bodies = BODIES.map((d, i) => {
      const fade = [];
      const track = (mat, base = 1) => {
        mat.transparent = true;
        mat.opacity = base;
        fade.push({ mat, base });
        return mat;
      };
      const group = new THREE.Group();
      if (d.kind === 'planet') {
        // El eje del planeta va inclinado hacia la cámara: así el anillo se abre (antes se veía de canto).
        group.rotation.set(d.lean, 0, d.tilt);
        const mesh = new THREE.Mesh(
          new THREE.SphereGeometry(d.r, 48, 32),
          track(new THREE.MeshStandardMaterial({ map: planetTexture(d.cols, d.bands, i * 77 + 13), roughness: 0.9 })),
        );
        this.spinners.push({ mesh, speed: 0.05 + i * 0.01 });
        group.add(mesh);
        if (d.ring) {
          [[1.45, 1.85, 0.7, 0xc8b48c], [1.95, 2.3, 0.4, 0xe0d0aa]].forEach(([from, to, opacity, color]) => {
            const ring = new THREE.Mesh(
              new THREE.RingGeometry(d.r * from, d.r * to, 96),
              track(new THREE.MeshBasicMaterial({ color, side: THREE.DoubleSide, depthWrite: false }), opacity),
            );
            ring.rotation.x = Math.PI / 2;
            group.add(ring);
          });
        }
      } else {
        const mat = new THREE.ShaderMaterial({
          vertexShader: SURFACE_VS,
          fragmentShader: SURFACE_FS,
          transparent: true,
          uniforms: { col: { value: new THREE.Color(d.color) }, time: { value: 0 }, uAlpha: { value: 1 } },
        });
        this.starMats.push(mat);
        fade.push({ mat, uniform: true });
        group.add(new THREE.Mesh(new THREE.SphereGeometry(d.r, 64, 48), mat));
        // Halos más contenidos que antes (3,4/7/15 × r): a corta distancia teñían toda la pantalla.
        [[2.6, 0.75], [5.2, 0.3], [9, 0.12]].forEach(([scale, opacity]) => {
          const sprite = new THREE.Sprite(
            track(new THREE.SpriteMaterial({ map: this.glow, color: d.color, blending: THREE.AdditiveBlending, depthWrite: false }), opacity),
          );
          sprite.scale.setScalar(d.r * scale);
          group.add(sprite);
        });
      }
      group.visible = false;
      this.flight.add(group);
      return { group, fade, alpha: -1 };
    });
    this.buildDest();
  }

  // Horizonte de destino: un planeta enorme que sube desde abajo mientras la nave frena.
  buildDest() {
    this.dest = new THREE.Group();
    this.scene.add(this.dest);
    const color = new THREE.Color(DEST.color);
    const surface = new THREE.ShaderMaterial({
      vertexShader: DEST_VS,
      fragmentShader: DEST_FS,
      transparent: true,

    const atmosphere = new THREE.ShaderMaterial({

      fragmentShader: ATMO_FS,
      transparent: true,

    this.dest.add(new THREE.Mesh(new THREE.SphereGeometry(DEST.r * 1.12, 128, 64), atmosphere));
  }

  // Coloca cuerpos y horizonte según el formato actual (se repite al redimensionar).
  layoutFlight() {
    placeBodies(this.camera.aspect).forEach((p, i) => this.bodies[i].group.position.set(p.x, p.y, p.z));

  }

  buildBackdrop() {
    const n = 2600;
    const pos = new Float32Array(n * 3);
    const size = new Float32Array(n);
    const bright = new Float32Array(n);
    for (let i = 0; i < n; i++) {
      const u = Math.random() * 2 - 1;
      const phi = Math.random() * Math.PI * 2;
      const r = Math.sqrt(1 - u * u);
      pos.set([r * Math.cos(phi) * 1200, u * 1200, r * Math.sin(phi) * 1200], i * 3);
      const k = Math.random();
      size[i] = 1.4 + k * k * 2.4;
      bright[i] = 0.35 + 0.65 * Math.random() ** 1.5;
    }
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    geo.setAttribute('aSize', new THREE.BufferAttribute(size, 1));
    geo.setAttribute('aBright', new THREE.BufferAttribute(bright, 1));
    this.backdropUniforms = { uAlpha: { value: 1 }, uPx: { value: this.renderer.getPixelRatio() } };
    const points = new THREE.Points(
      geo,
      new THREE.ShaderMaterial({
        uniforms: this.backdropUniforms,
        vertexShader: BACKDROP_VS,
        fragmentShader: BACKDROP_FS,
        transparent: true,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
      }),
    );
    points.frustumCulled = false;
    this.rig.add(points);
  }

  buildStars() {
    const n = this.stars.length;
    const sky = new Float32Array(n * 3);
    const space = new Float32Array(n * 3);
    const color = new Float32Array(n * 3);
    this.stars.forEach((s, i) => {
      sky.set(s.skyPos, i * 3);
      space.set(s.spacePos, i * 3);
      new THREE.Color(s.color).toArray(color, i * 3);
    });
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(sky, 3));
    geo.setAttribute('aSky', new THREE.BufferAttribute(sky, 3));
    geo.setAttribute('aSpace', new THREE.BufferAttribute(space, 3));
    geo.setAttribute('aColor', new THREE.BufferAttribute(color, 3));
    geo.setAttribute('aSize', new THREE.BufferAttribute(new Float32Array(n).fill(15), 1));
    this.uniforms = { uMix: { value: 0 }, uAlpha: { value: 1 }, uPx: { value: this.renderer.getPixelRatio() } };
    const points = new THREE.Points(
      geo,
      new THREE.ShaderMaterial({
        uniforms: this.uniforms,
        vertexShader: CATALOG_VS,
        fragmentShader: CATALOG_FS,
        transparent: true,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
      }),
    );
    points.frustumCulled = false;
    this.rig.add(points);
  }

  buildRing(radius, color) {
    const pts = Array.from({ length: 128 }, (_, i) => {
      const a = (i / 128) * Math.PI * 2;
      return new THREE.Vector3(Math.cos(a) * radius, 0, Math.sin(a) * radius);
    });
    const line = new THREE.LineLoop(
      new THREE.BufferGeometry().setFromPoints(pts),
      new THREE.LineBasicMaterial({ color, transparent: true, opacity: 0 }),

  buildLabels() {
    this.labels = new Map(
      this.stars.map((s) => {

  pinchDistance() {
    const [a, b] = [...this.pointers.values()];
    return Math.hypot(a.x - b.x, a.y - b.y);
  }

  bindEvents() {
    this.abort = new AbortController();
    const opts = { signal: this.abort.signal };
    const c = this.canvas;

    c.addEventListener('pointerdown', (e) => {
      if (this.fl.state === 'running') return this.skipFlight();
      if (!this.interactive) return undefined;
      this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
      if (this.pointers.size === 1) {
        this.drag = { x: e.clientX, y: e.clientY, x0: e.clientX, y0: e.clientY };
      } else {
        this.drag = null;
        this.pinch = this.pinchDistance();
      }
      c.setPointerCapture(e.pointerId);
      this.goto = null;
      return undefined;
    }, opts);

    c.addEventListener('pointermove', (e) => {
      if (!this.interactive) return;
      if (this.pointers.has(e.pointerId)) this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
      if (this.pointers.size === 2) {
        const d = this.pinchDistance();
        if (this.pinch) this.zoom(this.pinch / d);
        this.pinch = d;
      } else if (this.drag) {
        this.view.yaw += (e.clientX - this.drag.x) * 0.005;
        this.view.pitch = Math.max(-1.5, Math.min(1.5, this.view.pitch + (e.clientY - this.drag.y) * 0.005));
        this.drag.x = e.clientX;
        this.drag.y = e.clientY;
      } else if (e.pointerType === 'mouse') {
        this.hoveredId = this.nearest(e, 26)?.id ?? null;
        c.style.cursor = this.hoveredId ? 'pointer' : 'grab';
      }
    }, opts);

    const release = (e) => {
      const drag = this.drag;
      this.pointers.delete(e.pointerId);
      this.pinch = null;
      this.drag = null;
      if (e.type === 'pointerup' && drag && this.pointers.size === 0 && Math.hypot(e.clientX - drag.x0, e.clientY - drag.y0) < 6) {
        const star = this.nearest(e, e.pointerType === 'touch' ? 40 : 26);
        if (star) this.onSelect(star.id);
      }
    };
    c.addEventListener('pointerup', release, opts);
    c.addEventListener('pointercancel', release, opts);

    c.addEventListener('wheel', (e) => {
      e.preventDefault();
      if (this.interactive) this.zoom(Math.exp(e.deltaY * 0.001));
    }, { ...opts, passive: false });

    this.resizeObserver = new ResizeObserver(() => this.resize());
    this.resizeObserver.observe(c.parentElement);
  }

  resize() {
    const { clientWidth: w, clientHeight: h } = this.canvas.parentElement;
    if (!w || !h) return;
    this.renderer.setSize(w, h, false);
    this.camera.aspect = w / h;
    this.camera.updateProjectionMatrix();
    this.layoutFlight();
  }


  project(star) {
    const p = this.tmpA.fromArray(star.skyPos).lerp(this.tmpB.fromArray(star.spacePos), this.mix).add(this.anchor).project(this.camera);
    return { x: ((p.x + 1) / 2) * this.canvas.clientWidth, y: ((1 - p.y) / 2) * this.canvas.clientHeight, visible: p.z < 1 };
  }

  nearest(event, radius) {
    const rect = this.canvas.getBoundingClientRect();
    const x = event.clientX - rect.left;
    const y = event.clientY - rect.top;
    let best = null;
    let bestDist = radius;
    for (const s of this.stars) {
      const q = this.project(s);
      const d = q.visible ? Math.hypot(q.x - x, q.y - y) : Infinity;
      if (d < bestDist) {
        bestDist = d;
        best = s;
      }
    }
    return best;
  }


  updateCamera() {
    const { pitch, fov, dist } = this.view;
    const yaw = this.view.yaw + this.sway;
    const f = new THREE.Vector3(Math.cos(pitch) * Math.cos(yaw), Math.sin(pitch), -Math.cos(pitch) * Math.sin(yaw));
    const pos = f.clone().multiplyScalar(-dist * this.mix).add(this.anchor);
    this.camera.position.copy(pos);
    this.camera.lookAt(pos.clone().addScaledVector(f, SKY_RADIUS * (1 - this.mix) + dist * this.mix));

    const base = (fov * (1 - this.mix) + 55 * this.mix + this.kick * (1 - this.mix)) * DEG;
    const a = this.camera.aspect;
    this.camera.fov = ((a < 1 ? 2 * Math.atan(Math.tan(base / 2) / a) : base) / DEG);
    this.camera.updateProjectionMatrix();
  }

  updateLabels() {
    const show = this.interactive && this.starAlpha > 0.9;
    this.labelLayer.classList.toggle('is-on', show);
    if (!show) return;
    const { clientWidth: w, clientHeight: h } = this.canvas;
    for (const s of this.stars) {
      const el = this.labels.get(s.id);
      const q = this.project(s);
      const onScreen = q.visible && q.x > 0 && q.y > 0 && q.x < w && q.y < h;
      el.style.display = onScreen ? 'block' : 'none';
      if (!onScreen) continue;
      el.style.transform = `translate(${q.x + 14}px, ${q.y - 8}px)`;
      el.classList.toggle('is-selected', s.id === this.selectedId);
      el.textContent = s.id === this.hoveredId ? `${s.name} · ${s.spectralType.split('-')[0]} · ${Math.round(s.frequencyHz)} Hz` : s.name;
      if (s.id === this.selectedId) this.ring.style.transform = `translate(${q.x}px, ${q.y}px)`;
    }
    const sel = this.stars.find((s) => s.id === this.selectedId);
    this.ring.style.display = sel && this.project(sel).visible ? 'block' : 'none';
  }

  tick() {
    this.raf = requestAnimationFrame(this.tick);
    const now = performance.now();
    const dt = Math.min((now - this.lastT) / 1000, 0.05);
    this.lastT = now;
    this.time += dt;
    this.runTweens(now);
    this.stepFlight(dt);
    this.kick += (FOV_KICK * this.speedFrac - this.kick) * Math.min(1, dt * 4);
    this.warp.update(this.speedFrac, this.anchor.z);
    if (this.arrive !== this.arriveApplied) {
      this.arriveApplied = this.arrive;
      this.onArrive?.(this.arrive);
    }
    // Balanceo suave y acotado (±2,3°) en lugar de girar sin parar: el horizonte no se va de cuadro
    // si la persona deja la bienvenida abierta.
    const swayTarget = this.drift && !this.exploring && !this.drag ? 1 : 0;
    this.swayK += (swayTarget - this.swayK) * Math.min(1, dt * 2.5);
    this.sway = Math.sin(this.time * 0.22) * 0.04 * this.swayK;

    const bodiesOn = this.updateBodies();
    if (bodiesOn) {
      this.spinners.forEach((s) => (s.mesh.rotation.y += dt * 2 * s.speed));
      this.starMats.forEach((m) => (m.uniforms.time.value = this.time * 2));
    }
    if (this.destAlpha !== this.appliedDestAlpha) this.applyDestAlpha(this.destAlpha);
    if (this.dest.visible) {
      // El horizonte va pegado a la cámara. Sube con la misma curva cúbica con la que frena la nave.
      const { y, z, rise, exit } = this.destBase;
      const lift = (1 - this.arrive) ** 3 * rise + this.destSink * exit;
      this.dest.position.set(this.anchor.x, this.anchor.y + y - lift, this.anchor.z + z);
      this.destSurface.uniforms.time.value = this.time;
    }
    this.uniforms.uMix.value = this.mix;
    this.uniforms.uAlpha.value = this.starAlpha;

    if (this.goto) {
      const dy = wrapAngle(this.goto.yaw - this.view.yaw);
      this.view.yaw += dy * 0.1;
      this.view.pitch += (this.goto.pitch - this.view.pitch) * 0.1;
      if (Math.abs(dy) < 0.002) this.goto = null;
    }
    this.rig.position.copy(this.anchor);
    this.updateCamera();
    this.equator.material.opacity = 0.5 * (1 - this.mix) * Math.max(0, this.starAlpha - STAR_ALPHA_FLIGHT) / (1 - STAR_ALPHA_FLIGHT);
    this.rings.forEach((r) => (r.material.opacity = 0.7 * this.mix * Math.max(0, this.starAlpha - STAR_ALPHA_FLIGHT) / (1 - STAR_ALPHA_FLIGHT)));
    this.renderer.render(this.scene, this.camera);
    this.updateLabels();
  }
}

        selectedId={selectedId}
        onSelect={select}
        onSettle={settle}
        onArrive={applyArrive}
        interactive={phase === 'ready'}
      />

      <div className="ui">
        <Header
          mode={mode}
          onModeChange={setMode}
          muted={player.muted}
          onToggleMuted={player.toggleMuted}
          logoPending={phase !== 'ready' && !docked}
        />
        <div className="ui__mid">
          <StarCatalog stars={stars} selectedId={selectedId} onSelect={select} />
          <StarDetail star={selected} open={detailOpen} />
          <div className="zoom">
            <button className="btn btn--icon" aria-label="Acercar" onClick={() => mapRef.current?.zoom(0.85)}>+</button>
            <button className="btn btn--icon" aria-label="Alejar" onClick={() => mapRef.current?.zoom(1.18)}>−</button>
            <button className="btn btn--icon" aria-label="Restablecer vista" onClick={() => mapRef.current?.resetView()}>⌂</button>
          </div>
          <p className="hint">Arrastra para orbitar · rueda para zoom · clic en una estrella para escucharla</p>
        </div>
        <PlayerBar
          star={selected}
          playing={player.playingId === selectedId}
          engine={player.engine}
          volume={player.volume}
          onVolume={player.setVolume}
          onToggle={() => (player.playingId ? stop() : play(selected))}
          onPrev={() => step(-1)}
          onNext={() => step(1)}
          detailOpen={detailOpen}
          onToggleDetail={() => setDetailOpen((o) => !o)}
        />
      </div>

      <div ref={arriveRef} className="arrive">
        {(phase === 'welcome' || phase === 'entering') && (
          <Welcome leaving={phase === 'entering'} armed={armed} onExplore={explore} onDock={onDock} />
        )}
      </div>
    </div>
  );
}
