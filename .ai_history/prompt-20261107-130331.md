export const clamp01 = (v) => Math.min(1, Math.max(0, v));

// Resuelve una cubic-bezier(x1, y1, x2, y2) igual que lo hace CSS.
export function cubicBezier(x1, y1, x2, y2) {
  const cx = 3 * x1;
  const bx = 3 * (x2 - x1) - cx;
  const ax = 1 - cx - bx;
  const cy = 3 * y1;
  const by = 3 * (y2 - y1) - cy;
  const ay = 1 - cy - by;
  const sampleX = (t) => ((ax * t + bx) * t + cx) * t;
  const sampleY = (t) => ((ay * t + by) * t + cy) * t;
  const slopeX = (t) => (3 * ax * t + 2 * bx) * t + cx;

  return (x) => {
    if (x <= 0) return 0;
    if (x >= 1) return 1;
    let t = x;
    for (let i = 0; i < 8; i += 1) {
      const err = sampleX(t) - x;
      if (Math.abs(err) < 1e-5) return sampleY(t);
      const d = slopeX(t);
      if (Math.abs(d) < 1e-6) break;
      t -= err / d;
    }
    let lo = 0;
    let hi = 1;
    t = x;
    for (let i = 0; i < 32; i += 1) {
      const err = sampleX(t) - x;
      if (Math.abs(err) < 1e-5) break;
      if (err > 0) hi = t;
      else lo = t;
      t = (lo + hi) / 2;
    }
    return sampleY(t);
  };
}

  exit: cubicBezier(0.3, 0, 0.8, 0.15),
  inOut: cubicBezier(0.65, 0, 0.35, 1),
};

// Estelas de velocidad: partículas fijas en el mundo que la cámara va "devorando".
// El largo del trazo es proporcional a la velocidad, así que al frenar las estelas
// se encogen hasta volver a ser puntos: el espectador "cae" desde el vuelo hacia
// el cielo quieto sin que nada desaparezca de golpe.

const COUNT = 720;
const NEAR = 140; // distancia mínima a la cámara antes de reciclar la partícula
const FAR = 2600; // profundidad máxima de aparición
const FADE_NEAR = 260;
const FADE_FAR = 900;
const TRAIL_MAX = 46; // largo máximo del trazo (unidades de mundo) a velocidad máxima
const PALETTE = [
  [0.72, 0.84, 1.0],
  [0.9, 0.96, 1.0],
  [0.5, 0.92, 0.86], // guiño al teal de la marca
  [1.0, 0.88, 0.7], // guiño a las estrellas cálidas del vuelo
];

const smooth = (v) => {
  const t = clamp01(v);
  return t * t * (3 - 2 * t);
};

export class WarpField {
  constructor() {
    this.pos = new Float32Array(COUNT * 3);
    this.base = new Float32Array(COUNT * 3);
    this.positions = new Float32Array(COUNT * 6);
    this.colors = new Float32Array(COUNT * 6);

    for (let i = 0; i < COUNT; i += 1) {
      const c = PALETTE[Math.floor(Math.random() ** 1.6 * PALETTE.length)];
      const b = 0.55 + 0.45 * Math.random();
      this.base.set([c[0] * b, c[1] * b, c[2] * b], i * 3);
    }

    const geometry = new THREE.BufferGeometry();
    this.positionAttr = new THREE.BufferAttribute(this.positions, 3);
    this.colorAttr = new THREE.BufferAttribute(this.colors, 3);
    geometry.setAttribute('position', this.positionAttr);
    geometry.setAttribute('color', this.colorAttr);

    this.material = new THREE.LineBasicMaterial({
      vertexColors: true,
      transparent: true,
      opacity: 0,
      blending: THREE.AdditiveBlending,
      depthWrite: false,
    });
    this.object = new THREE.LineSegments(geometry, this.material);
    this.object.frustumCulled = false;
    this.object.visible = false;
  }

  place(i, z) {
    const r = 70 + 430 * Math.random() ** 0.8;
    const a = Math.random() * Math.PI * 2;
    this.pos[i * 3] = Math.cos(a) * r;
    this.pos[i * 3 + 1] = Math.sin(a) * r;
    this.pos[i * 3 + 2] = z;
  }

  reset(camZ) {
    for (let i = 0; i < COUNT; i += 1) this.place(i, camZ - NEAR - Math.random() * FAR);
  }

  update(speedFrac, camZ) {
    const strength = clamp01(speedFrac * 1.5) * 0.9;
    this.material.opacity = strength;
    const on = strength > 0.01;
    this.object.visible = on;
    if (!on) return;

    const trail = TRAIL_MAX * speedFrac;
    for (let i = 0; i < COUNT; i += 1) {
      let z = this.pos[i * 3 + 2];
      let d = camZ - z;
      if (d < NEAR) {
        z = camZ - FAR - Math.random() * 400;
        this.place(i, z);
        d = camZ - z;
      }
      const x = this.pos[i * 3];
      const y = this.pos[i * 3 + 1];
      const fade = smooth((d - NEAR) / FADE_NEAR) * (1 - smooth((d - FADE_FAR) / (FAR - FADE_FAR)));
      const j = i * 6;
      this.positions[j] = x;
      this.positions[j + 1] = y;
      this.positions[j + 2] = z;
      this.positions[j + 3] = x;
      this.positions[j + 4] = y;
      this.positions[j + 5] = z - trail;
      this.colors[j] = this.base[i * 3] * fade;
      this.colors[j + 1] = this.base[i * 3 + 1] * fade;
      this.colors[j + 2] = this.base[i * 3 + 2] * fade;
      this.colors[j + 3] = 0;
      this.colors[j + 4] = 0;
      this.colors[j + 5] = 0;
    }
    this.positionAttr.needsUpdate = true;
    this.colorAttr.needsUpdate = true;
  }
}

function planetTexture(cols, bands, seed) {
  const c = document.createElement('canvas');
  c.width = 512;
  c.height = 256;
  const g = c.getContext('2d');
  let s = seed;
  const rnd = () => (s = (s * 9301 + 49297) % 233280) / 233280;
  for (let i = 0; i < bands; i++) {
    g.fillStyle = cols[Math.floor(rnd() * cols.length)];

          sprite.scale.setScalar(d.r * scale);
          mesh.add(sprite);
        });
      }
      mesh.position.set(d.x, d.y, d.z * Z_SCALE);
      this.flight.add(mesh);
    });
  }

  buildBackdrop() {
    const n = 2600;
    const pos = new Float32Array(n * 3);
    const size = new Float32Array(n);
    const bright = new Float32Array(n);
    for (let i = 0; i < n; i++) {
      const u = Math.random() * 2 - 1;
      const phi = Math.random() * Math.PI * 2;
      const r = Math.sqrt(1 - u * u);
      pos.set([r * Math.cos(phi) * 1200, u * 1200, r * Math.sin(phi) * 1200], i * 3);
      const k = Math.random();
      size[i] = 1.4 + k * k * 2.4;
      bright[i] = 0.35 + 0.65 * Math.random() ** 1.5;
    }
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    geo.setAttribute('aSize', new THREE.BufferAttribute(size, 1));
    geo.setAttribute('aBright', new THREE.BufferAttribute(bright, 1));
    this.backdropUniforms = { uAlpha: { value: 1 }, uPx: { value: this.renderer.getPixelRatio() } };
    const points = new THREE.Points(
      geo,
      new THREE.ShaderMaterial({
        uniforms: this.backdropUniforms,
        vertexShader: BACKDROP_VS,
        fragmentShader: BACKDROP_FS,
        transparent: true,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
      }),
    );
    points.frustumCulled = false;
    this.rig.add(points);
  }

  buildStars() {
    const n = this.stars.length;
    const sky = new Float32Array(n * 3);
    const space = new Float32Array(n * 3);
    const color = new Float32Array(n * 3);
    this.stars.forEach((s, i) => {
      sky.set(s.skyPos, i * 3);
      space.set(s.spacePos, i * 3);
      new THREE.Color(s.color).toArray(color, i * 3);
    });
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(sky, 3));
    geo.setAttribute('aSky', new THREE.BufferAttribute(sky, 3));
    geo.setAttribute('aSpace', new THREE.BufferAttribute(space, 3));
    geo.setAttribute('aColor', new THREE.BufferAttribute(color, 3));
    geo.setAttribute('aSize', new THREE.BufferAttribute(new Float32Array(n).fill(15), 1));
    this.uniforms = { uMix: { value: 0 }, uAlpha: { value: 1 }, uPx: { value: this.renderer.getPixelRatio() } };
    const points = new THREE.Points(
      geo,
      new THREE.ShaderMaterial({
        uniforms: this.uniforms,
        vertexShader: CATALOG_VS,
        fragmentShader: CATALOG_FS,
        transparent: true,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
      }),
    );
    points.frustumCulled = false;
    this.rig.add(points);
  }

  buildRing(radius, color) {
    const pts = Array.from({ length: 128 }, (_, i) => {
      const a = (i / 128) * Math.PI * 2;
      return new THREE.Vector3(Math.cos(a) * radius, 0, Math.sin(a) * radius);
    });
    const line = new THREE.LineLoop(
      new THREE.BufferGeometry().setFromPoints(pts),
      new THREE.LineBasicMaterial({ color, transparent: true, opacity: 0 }),
    );
    this.rig.add(line);
    return line;
  }

  buildLabels() {
    this.labels = new Map(
      this.stars.map((s) => {
        const el = document.createElement('div');
        el.className = 'star-label';
        el.textContent = s.name;
        this.labelLayer.appendChild(el);
        return [s.id, el];
      }),
    );
    this.ring = document.createElement('div');
    this.ring.className = 'star-ring';
    this.labelLayer.appendChild(this.ring);
  }


  pinchDistance() {
    const [a, b] = [...this.pointers.values()];
    return Math.hypot(a.x - b.x, a.y - b.y);
  }

  bindEvents() {
    this.abort = new AbortController();
    const opts = { signal: this.abort.signal };
    const c = this.canvas;

    c.addEventListener('pointerdown', (e) => {
      if (this.fl.state === 'running') return this.skipFlight();
      if (!this.interactive) return undefined;
      this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
      if (this.pointers.size === 1) {
        this.drag = { x: e.clientX, y: e.clientY, x0: e.clientX, y0: e.clientY };
      } else {
        this.drag = null;
        this.pinch = this.pinchDistance();
      }
      c.setPointerCapture(e.pointerId);
      this.goto = null;
      return undefined;
    }, opts);

    c.addEventListener('pointermove', (e) => {
      if (!this.interactive) return;
      if (this.pointers.has(e.pointerId)) this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
      if (this.pointers.size === 2) {
        const d = this.pinchDistance();
        if (this.pinch) this.zoom(this.pinch / d);
        this.pinch = d;
      } else if (this.drag) {
        this.view.yaw += (e.clientX - this.drag.x) * 0.005;
        this.view.pitch = Math.max(-1.5, Math.min(1.5, this.view.pitch + (e.clientY - this.drag.y) * 0.005));
        this.drag.x = e.clientX;
        this.drag.y = e.clientY;
      } else if (e.pointerType === 'mouse') {
        this.hoveredId = this.nearest(e, 26)?.id ?? null;
        c.style.cursor = this.hoveredId ? 'pointer' : 'grab';
      }
    }, opts);

    const release = (e) => {
      const drag = this.drag;
      this.pointers.delete(e.pointerId);
      this.pinch = null;
      this.drag = null;
      if (e.type === 'pointerup' && drag && this.pointers.size === 0 && Math.hypot(e.clientX - drag.x0, e.clientY - drag.y0) < 6) {
        const star = this.nearest(e, e.pointerType === 'touch' ? 40 : 26);
        if (star) this.onSelect(star.id);
      }
    };
    c.addEventListener('pointerup', release, opts);
    c.addEventListener('pointercancel', release, opts);

    c.addEventListener('wheel', (e) => {
      e.preventDefault();
      if (this.interactive) this.zoom(Math.exp(e.deltaY * 0.001));
    }, { ...opts, passive: false });

    this.resizeObserver = new ResizeObserver(() => this.resize());
    this.resizeObserver.observe(c.parentElement);
  }

  resize() {
    const { clientWidth: w, clientHeight: h } = this.canvas.parentElement;
    if (!w || !h) return;
    this.renderer.setSize(w, h, false);
    this.camera.aspect = w / h;
    this.camera.updateProjectionMatrix();
  }


  project(star) {
    const p = this.tmpA.fromArray(star.skyPos).lerp(this.tmpB.fromArray(star.spacePos), this.mix).add(this.anchor).project(this.camera);
    return { x: ((p.x + 1) / 2) * this.canvas.clientWidth, y: ((1 - p.y) / 2) * this.canvas.clientHeight, visible: p.z < 1 };
  }

  nearest(event, radius) {
    const rect = this.canvas.getBoundingClientRect();
    const x = event.clientX - rect.left;
    const y = event.clientY - rect.top;
    let best = null;
    let bestDist = radius;
    for (const s of this.stars) {
      const q = this.project(s);
      const d = q.visible ? Math.hypot(q.x - x, q.y - y) : Infinity;
      if (d < bestDist) {
        bestDist = d;
        best = s;
      }
    }
    return best;
  }


  updateCamera() {
    const { yaw, pitch, fov, dist } = this.view;
    const f = new THREE.Vector3(Math.cos(pitch) * Math.cos(yaw), Math.sin(pitch), -Math.cos(pitch) * Math.sin(yaw));
    const pos = f.clone().multiplyScalar(-dist * this.mix).add(this.anchor);
    this.camera.position.copy(pos);
    this.camera.lookAt(pos.clone().addScaledVector(f, SKY_RADIUS * (1 - this.mix) + dist * this.mix));

    const base = (fov * (1 - this.mix) + 55 * this.mix + this.kick * (1 - this.mix)) * DEG;
    const a = this.camera.aspect;
    this.camera.fov = ((a < 1 ? 2 * Math.atan(Math.tan(base / 2) / a) : base) / DEG);
    this.camera.updateProjectionMatrix();
  }

  updateLabels() {
    const show = this.interactive && this.starAlpha > 0.9;
    this.labelLayer.classList.toggle('is-on', show);
    if (!show) return;
    const { clientWidth: w, clientHeight: h } = this.canvas;
    for (const s of this.stars) {
      const el = this.labels.get(s.id);
      const q = this.project(s);
      const onScreen = q.visible && q.x > 0 && q.y > 0 && q.x < w && q.y < h;
      el.style.display = onScreen ? 'block' : 'none';
      if (!onScreen) continue;
      el.style.transform = `translate(${q.x + 14}px, ${q.y - 8}px)`;
      el.classList.toggle('is-selected', s.id === this.selectedId);
      el.textContent = s.id === this.hoveredId ? `${s.name} · ${s.spectralType.split('-')[0]} · ${Math.round(s.frequencyHz)} Hz` : s.name;
      if (s.id === this.selectedId) this.ring.style.transform = `translate(${q.x}px, ${q.y}px)`;
    }
    const sel = this.stars.find((s) => s.id === this.selectedId);
    this.ring.style.display = sel && this.project(sel).visible ? 'block' : 'none';
  }

  tick() {
    this.raf = requestAnimationFrame(this.tick);
    const now = performance.now();
    const dt = Math.min((now - this.lastT) / 1000, 0.05);
    this.lastT = now;
    this.time += dt;
    this.runTweens(now);
    this.stepFlight(dt);
    this.kick += (FOV_KICK * this.speedFrac - this.kick) * Math.min(1, dt * 4);
    this.warp.update(this.speedFrac, this.anchor.z);
    if (this.arrive !== this.arriveApplied) {
      this.arriveApplied = this.arrive;
      this.onArrive?.(this.arrive);
    }
    if (this.drift && !this.exploring && !this.drag) this.view.yaw += dt * 0.02;

    if (this.flightAlpha !== this.appliedFlightAlpha) this.applyFlightAlpha(this.flightAlpha);
    if (this.flight.visible) {
      this.spinners.forEach((s) => (s.mesh.rotation.y += dt * 2 * s.speed));
      this.starMats.forEach((m) => (m.uniforms.time.value = this.time * 2));
    }
    this.uniforms.uMix.value = this.mix;
    this.uniforms.uAlpha.value = this.starAlpha;

    if (this.goto) {
      const dy = wrapAngle(this.goto.yaw - this.view.yaw);
      this.view.yaw += dy * 0.1;
      this.view.pitch += (this.goto.pitch - this.view.pitch) * 0.1;
      if (Math.abs(dy) < 0.002) this.goto = null;
    }
    this.rig.position.copy(this.anchor);
    this.updateCamera();
    this.equator.material.opacity = 0.5 * (1 - this.mix) * Math.max(0, this.starAlpha - STAR_ALPHA_FLIGHT) / (1 - STAR_ALPHA_FLIGHT);
    this.rings.forEach((r) => (r.material.opacity = 0.7 * this.mix * Math.max(0, this.starAlpha - STAR_ALPHA_FLIGHT) / (1 - STAR_ALPHA_FLIGHT)));
    this.renderer.render(this.scene, this.camera);
    this.updateLabels();
  }
}

import { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import catalog from './data/stars.json';
import { enrichStar } from './domain/starMath.js';
import { useStarPlayer } from './hooks/useStarPlayer.js';
import { EASE, clamp01 } from './motion.js';
import Welcome from './components/Welcome.jsx';
import Header from './components/Header.jsx';
import StarCanvas from './components/StarCanvas.jsx';
import StarCatalog from './components/StarCatalog.jsx';
import StarDetail from './components/StarDetail.jsx';
import PlayerBar from './components/PlayerBar.jsx';



const SEEN_KEY = 'astrowave:intro-seen';

// Coreografía de la llegada. p (0 → 1) es el avance de la frenada de la cámara.
// Cada elemento de la bienvenida toma un tramo de p: el logo resuelve primero (cuando
// la nave casi se detuvo y la estela se encoge) y el botón al final. Un solo reloj para
// el 3D y el DOM: lo que se ve en el canvas y en la interfaz es el mismo movimiento.
const REVEAL = { glow: [0, 0.8], logo: [0.08, 0.55], lead: [0.32, 0.78], sub: [0.46, 0.9], cta: [0.6, 1] };
const ARM_AT = 0.72; // el botón solo se activa cuando ya es visible
const reveal = (p, [from, to]) => EASE.standard(clamp01((p - from) / (to - from)));
const prefersReducedMotion = () => window.matchMedia('(prefers-reduced-motion: reduce)').matches;

function introAlreadySeen() {
  if (new URLSearchParams(window.location.search).has('intro')) return false;
  try {
    return localStorage.getItem(SEEN_KEY) === '1';
  } catch {
    return false;
  }
}

export default function App() {
  const stars = useMemo(() => catalog.map(enrichStar), []);
  const [intro] = useState(() => (prefersReducedMotion() || introAlreadySeen() ? 'idle' : 'flight'));
  const [phase, setPhase] = useState(intro === 'flight' ? 'flight' : 'welcome');
  const [mode, setMode] = useState('sky');
  const [selectedId, setSelectedId] = useState(stars[0].id);
  const [detailOpen, setDetailOpen] = useState(false);
  const mapRef = useRef(null);
  const arriveRef = useRef(null);
  const armedRef = useRef(false);
  const [armed, setArmed] = useState(false);
  const [docked, setDocked] = useState(false);
  const player = useStarPlayer();
  const { play, stop, unlock } = player;

  const selected = stars.find((s) => s.id === selectedId);

  // Escribe el progreso como variables CSS en un único contenedor (sin re-render de React).
  const applyArrive = useCallback((p) => {
    const el = arriveRef.current;
    if (!el) return;
    Object.entries(REVEAL).forEach(([name, range]) => {
      el.style.setProperty(`--a-${name}`, reveal(p, range).toFixed(4));
    });
    if (!armedRef.current && p >= ARM_AT) {
      armedRef.current = true;
      setArmed(true);
    }
  }, []);

  const onDock = useCallback(() => setDocked(true), []);

  // Red de seguridad: si por algo el 3D no reporta la llegada, la bienvenida igual aparece.
  useEffect(() => {
    if (phase !== 'welcome') return undefined;
    const id = window.setTimeout(() => applyArrive(1), 4500);
    return () => window.clearTimeout(id);
  }, [phase, applyArrive]);


  const settle = useCallback(() => {
    try {
      localStorage.setItem(SEEN_KEY, '1');
    } catch {

    }
    setPhase((p) => (p === 'flight' ? 'welcome' : p));
  }, []);


  useEffect(() => {
    if (phase !== 'flight') return undefined;
    const onKey = (e) => {
      if (['Escape', ' ', 'Enter'].includes(e.key)) {
        e.preventDefault();
        mapRef.current?.skip();
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [phase]);

  const explore = () => {
    if (phase !== 'welcome') return;
    unlock().catch(() => {});
    applyArrive(1);
    setPhase('entering');
    mapRef.current?.explore(() => setPhase('ready'));
  };

  const select = useCallback(
    (id) => {
      setSelectedId(id);
      play(stars.find((s) => s.id === id));
    },
    [play, stars],
  );

  const step = (delta) => {
    const i = stars.findIndex((s) => s.id === selectedId);
    select(stars[(i + delta + stars.length) % stars.length].id);
  };

  return (
    <div className="app" data-phase={phase}>
      <StarCanvas
        ref={mapRef}
        stars={stars}
        intro={intro}
        mode={mode}
        selectedId={selectedId}
        onSelect={select}
        onSettle={settle}
        onArrive={applyArrive}
        interactive={phase === 'ready'}
      />

      <div className="ui">
        <Header
          mode={mode}
          onModeChange={setMode}
          muted={player.muted}
          onToggleMuted={player.toggleMuted}
          logoPending={phase !== 'ready' && !docked}
        />
        <div className="ui__mid">
          <StarCatalog stars={stars} selectedId={selectedId} onSelect={select} />
          <StarDetail star={selected} open={detailOpen} />
          <div className="zoom">
            <button className="btn btn--icon" aria-label="Acercar" onClick={() => mapRef.current?.zoom(0.85)}>+</button>
            <button className="btn btn--icon" aria-label="Alejar" onClick={() => mapRef.current?.zoom(1.18)}>−</button>
            <button className="btn btn--icon" aria-label="Restablecer vista" onClick={() => mapRef.current?.resetView()}>⌂</button>
          </div>
          <p className="hint">Arrastra para orbitar · rueda para zoom · clic en una estrella para escucharla</p>
        </div>
        <PlayerBar
          star={selected}
          playing={player.playingId === selectedId}
          engine={player.engine}
          volume={player.volume}
          onVolume={player.setVolume}
          onToggle={() => (player.playingId ? stop() : play(selected))}
          onPrev={() => step(-1)}
          onNext={() => step(1)}
          detailOpen={detailOpen}
          onToggleDetail={() => setDetailOpen((o) => !o)}
        />
      </div>

      {phase === 'flight' && (
        <button className="btn skip" onClick={() => mapRef.current?.skip()}>Saltar intro</button>
      )}
      <div ref={arriveRef} className="arrive">
        {(phase === 'welcome' || phase === 'entering') && (
          <Welcome leaving={phase === 'entering'} armed={armed} onExplore={explore} onDock={onDock} />
        )}
      </div>
    </div>
  );
}
