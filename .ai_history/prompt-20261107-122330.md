// Viaje por el espacio de la introducción (portado de space_video.html).
// Escena autónoma con su propio renderer: se destruye al terminar para liberar la GPU.
import * as THREE from 'three';
import { LIGHT_SCALE, OUTPUT_COLOR_SPACE } from './legacyColor.js';

const VMAX = 400;
const END_T = 22; // "tiempo de película" (2x el real): ~11 s. Las escenas vacías finales se recortan.
const speedAt = (t) => {
  if (t < 4) return VMAX * (t / 4) ** 2;
  if (t < 14) return VMAX;
  if (t < 20) return VMAX * (1 - (t - 14) / 6) ** 2;
  return 0;
};

const planet = (z, x, y, r, cols, bands, ring) => ({ kind: 'planet', z, x, y, r, cols, bands, ring });
const star = (z, x, y, r, color) => ({ kind: 'star', z, x, y, r, color });
const DEFS = [
  planet(-450, -75, 15, 30, ['#c9774a', '#a85a35', '#e0a070'], 14),
  star(-800, 120, -20, 45, 0xffd27a),
  planet(-1100, 105, -30, 50, ['#d9c08a', '#bfa06a', '#efe0b8', '#a88a55'], 22, true),
  star(-1500, -130, 40, 70, 0xa8c8ff),
  star(-1950, 95, 55, 40, 0xff9a45),
  star(-2350, -100, -45, 85, 0xff5a2a),
  planet(-2750, 90, 25, 28, ['#8a8a8a', '#6d6d6d', '#aaaaaa'], 10),
  star(-3150, -120, 10, 55, 0xfff4e0),
  planet(-3500, -100, -10, 65, ['#7ac2b0', '#4f9a8c', '#a8e0d0', '#2f7468'], 18),
  star(-3900, 130, -35, 95, 0xffc060),
  planet(-4250, 115, 15, 60, ['#b5543c', '#8a3a2a', '#d98a5a', '#e8b080'], 16, true),
  star(-4600, -90, 20, 40, 0xff8a50),
];

const STAR_VS = 'varying vec3 vN;varying vec3 vP;void main(){vN=normalize(normalMatrix*normal);vP=position;gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.0);}';
const STAR_FS = `precision highp float;varying vec3 vN;varying vec3 vP;uniform vec3 col;uniform float time;
float h(vec3 p){p=fract(p*0.3183099+.1);p*=17.0;return fract(p.x*p.y*p.z*(p.x+p.y+p.z));}
float n(vec3 x){vec3 i=floor(x),f=fract(x);f=f*f*(3.0-2.0*f);
return mix(mix(mix(h(i),h(i+vec3(1,0,0)),f.x),mix(h(i+vec3(0,1,0)),h(i+vec3(1,1,0)),f.x),f.y),
mix(mix(h(i+vec3(0,0,1)),h(i+vec3(1,0,1)),f.x),mix(h(i+vec3(0,1,1)),h(i+vec3(1,1,1)),f.x),f.y),f.z);}
void main(){
 vec3 q=normalize(vP)*3.0;
 float g=n(q*2.0+time*.15)*.5+n(q*5.0-time*.2)*.3+n(q*11.0+time*.3)*.2;
 float d=clamp(dot(normalize(vN),vec3(0,0,1)),0.0,1.0);
 float limb=pow(d,.45);
 vec3 c=mix(col*.55,col*1.35+vec3(.25),g);
 c*=mix(.35,1.25,limb);
 c+=col*pow(1.0-d,3.0)*.25;
 gl_FragColor=vec4(c,1.0);}`;

function planetTexture(cols, bands, seed) {
  const c = document.createElement('canvas');
  c.width = 512;
  c.height = 256;
  const g = c.getContext('2d');
  let s = seed;
  const rnd = () => (s = (s * 9301 + 49297) % 233280) / 233280;
  for (let i = 0; i < bands; i++) {
    g.fillStyle = cols[Math.floor(rnd() * cols.length)];
    g.fillRect(0, (i * 256) / bands, 512, 256 / bands + 1);
  }
  for (let i = 0; i < 260; i++) {
    g.fillStyle = `rgba(${rnd() > 0.5 ? '255,255,255' : '0,0,0'},${rnd() * 0.12})`;
    g.beginPath();
    g.ellipse(rnd() * 512, rnd() * 256, rnd() * 50 + 6, rnd() * 8 + 2, 0, 0, 7);
    g.fill();
  }
  return new THREE.CanvasTexture(c);
}

function glowTexture() {
  const c = document.createElement('canvas');
  c.width = c.height = 128;
  const g = c.getContext('2d');
  const gr = g.createRadialGradient(64, 64, 0, 64, 64, 64);
  gr.addColorStop(0, 'rgba(255,255,255,1)');
  gr.addColorStop(0.25, 'rgba(255,255,255,.45)');
  gr.addColorStop(1, 'rgba(255,255,255,0)');
  g.fillStyle = gr;
  g.fillRect(0, 0, 128, 128);
  return new THREE.CanvasTexture(c);
}

export class FlightScene {
  constructor({ canvas, onEnd }) {
    this.onEnd = onEnd;
    this.t = 0;
    this.camZ = 0;
    this.ended = false;
    this.disposed = false;
    this.starMaterials = [];

    this.renderer = new THREE.WebGLRenderer({ canvas, antialias: true });
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.outputColorSpace = OUTPUT_COLOR_SPACE;
    this.renderer.setClearColor(0x000000, 1);
    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(70, 1, 0.5, 4000);

    this.scene.add(new THREE.AmbientLight(0x303850, 0.6 * LIGHT_SCALE));
    const sun = new THREE.DirectionalLight(0xfff0dd, 1.6 * LIGHT_SCALE);
    sun.position.set(1, 0.6, 0.8);
    this.scene.add(sun);

    this.glow = glowTexture();
    this.objects = DEFS.map((d, i) => this.buildObject(d, i));

    this.abort = new AbortController();
    window.addEventListener('resize', () => this.resize(), { signal: this.abort.signal });
    this.resize();
    this.last = performance.now();
    this.loop = this.loop.bind(this);
    this.raf = requestAnimationFrame(this.loop);
  }

  buildObject(d, i) {
    let mesh;
    if (d.kind === 'planet') {
      mesh = new THREE.Mesh(
        new THREE.SphereGeometry(d.r, 48, 32),
        new THREE.MeshStandardMaterial({ map: planetTexture(d.cols, d.bands, i * 77 + 13), roughness: 0.9 }),
      );
      if (d.ring) {
        const ring = new THREE.Mesh(
          new THREE.RingGeometry(d.r * 1.4, d.r * 2.2, 64),
          new THREE.MeshBasicMaterial({ color: 0xc8b48c, side: THREE.DoubleSide, transparent: true, opacity: 0.55 }),
        );
        ring.rotation.x = Math.PI / 2.4;
        mesh.add(ring);
      }
    } else {
      const material = new THREE.ShaderMaterial({
        vertexShader: STAR_VS,
        fragmentShader: STAR_FS,
        uniforms: { col: { value: new THREE.Color(d.color) }, time: { value: 0 } },
      });
      this.starMaterials.push(material);
      mesh = new THREE.Mesh(new THREE.SphereGeometry(d.r, 64, 48), material);
      [[3.4, 0.9], [7, 0.5], [15, 0.22]].forEach(([scale, opacity]) => {
        const sprite = new THREE.Sprite(
          new THREE.SpriteMaterial({ map: this.glow, color: d.color, blending: THREE.AdditiveBlending, transparent: true, opacity, depthWrite: false }),
        );
        sprite.scale.setScalar(d.r * scale);
        mesh.add(sprite);
      });
    }
    mesh.position.set(d.x, d.y, d.z);
    this.scene.add(mesh);
    return mesh;
  }

  resize() {
    this.renderer.setSize(window.innerWidth, window.innerHeight, false);
    this.camera.aspect = window.innerWidth / window.innerHeight;
    this.camera.updateProjectionMatrix();
  }

  loop(now) {
    this.raf = requestAnimationFrame(this.loop);
    // El tiempo se acumula: si la pestaña se oculta, el vuelo se pausa en vez de saltarse.
    const dt = Math.min((now - this.last) / 1000, 0.05) * 2;
    this.last = now;
    this.t += dt;
    const t = this.t;
    const v = speedAt(t);
    this.camZ -= v * dt;

    const cam = this.camera;
    cam.position.set(Math.sin(t * 0.6) * 1.2, Math.cos(t * 0.45) * 0.8, this.camZ);
    cam.rotation.z = Math.sin(t * 0.3) * 0.02 * (v / VMAX);
    cam.lookAt(cam.position.x * 0.5, cam.position.y * 0.5, this.camZ - 100);

    this.objects.forEach((o, i) => {
      if (DEFS[i].kind === 'planet') o.rotation.y += dt * (0.05 + i * 0.01);
    });
    this.starMaterials.forEach((m) => (m.uniforms.time.value = t));
    this.renderer.render(this.scene, cam);

    if (t >= END_T && !this.ended) {
      this.ended = true;
      this.onEnd?.();
    }
  }

  dispose() {
    this.disposed = true;
    cancelAnimationFrame(this.raf);
    this.abort.abort();
    this.scene.traverse((o) => {
      o.geometry?.dispose();
      if (o.material) {
        o.material.map?.dispose();
        o.material.dispose();
      }
    });
    this.glow.dispose();
    this.renderer.dispose();
    this.renderer.forceContextLoss(); // libera el contexto WebGL de inmediato
  }
}

// Motor 3D del mapa, independiente de React. Modos: 'sky' (esfera celeste) y 'space' (vecindad 3D).
// Transición: cada estrella interpola entre su posición en la esfera y su posición espacial en el shader.
// Además expone la coreografía de entrada: enterIntroPose() -> revealBackdrop() -> explore().
import * as THREE from 'three';
import { OUTPUT_COLOR_SPACE } from './legacyColor.js';
import { SKY_RADIUS } from '../domain/starMath.js';

const DEFAULT_VIEW = { yaw: (80 * Math.PI) / 180, pitch: 0, fov: 60, dist: 220 };
const INTRO_VIEW = { yaw: 1.2, pitch: 0.45, fov: 60, dist: 520 };
const LIMITS = { fov: [25, 90], dist: [60, 700] };
const PICK_RADIUS_PX = 26;
const clamp = (v, [lo, hi]) => Math.min(hi, Math.max(lo, v));
const wrapAngle = (a) => ((((a + Math.PI) % (2 * Math.PI)) + 2 * Math.PI) % (2 * Math.PI)) - Math.PI;
const easeInOut = (t) => (t < 0.5 ? 2 * t * t : 1 - (-2 * t + 2) ** 2 / 2);

const VERTEX = `
attribute vec3 aSky; attribute vec3 aSpace; attribute vec3 aColor; attribute float aSize;
uniform float uMix; uniform float uPx; varying vec3 vColor;
void main(){
  vColor = aColor;
  vec3 p = mix(aSky, aSpace, uMix);
  gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);
  gl_PointSize = aSize * uPx;
}`;
const FRAGMENT = `
varying vec3 vColor; uniform float uAlpha;
void main(){
  float d = length(gl_PointCoord - 0.5) * 2.0;
  if (d > 1.0) discard;
  float a = smoothstep(1.0, 0.0, d);
  gl_FragColor = vec4(vColor * (0.4 + 0.9 * a * a), a * uAlpha);
}`;

export class SkyScene {
  constructor({ canvas, labelLayer, stars, onSelect, reducedMotion = false }) {
    Object.assign(this, { canvas, labelLayer, stars, onSelect, reducedMotion });
    this.view = { ...DEFAULT_VIEW };
    this.mix = 0;
    this.mixTarget = 0;
    this.starAlpha = 1;
    this.backdropAlpha = 1;
    this.drift = false;
    this.interactive = true;
    this.tweens = [];
    this.goto = null;
    this.selectedId = null;
    this.hoveredId = null;
    this.drag = null;
    this.tmpA = new THREE.Vector3();
    this.tmpB = new THREE.Vector3();

    this.renderer = new THREE.WebGLRenderer({ canvas, antialias: true });
    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    this.renderer.outputColorSpace = OUTPUT_COLOR_SPACE;
    this.scene = new THREE.Scene();
    this.camera = new THREE.PerspectiveCamera(60, 1, 0.1, 4000);

    this.buildStars();
    this.buildBackdrop();
    this.equator = this.buildRing(SKY_RADIUS, 0x3a3f66);
    this.rings = [5, 25, 100, 250].map((pc) => this.buildRing(6 * Math.sqrt(pc), 0x2a2e4d));
    this.buildLabels();
    this.bindEvents();
    this.resize();
    this.lastT = performance.now();
    this.tick = this.tick.bind(this);
    this.raf = requestAnimationFrame(this.tick);
  }

  // ---- API pública ----
  setMode(mode, ms = 1400) {
    this.mixTarget = mode === 'space' ? 1 : 0;
    this.tween(this, 'mix', this.mixTarget, ms);
  }

  setSelected(id) {
    this.selectedId = id;
    const star = this.stars.find((s) => s.id === id);
    if (star && this.mixTarget === 0 && this.interactive) {
      const [x, y, z] = star.skyPos;
      this.goto = { yaw: Math.atan2(-z, x), pitch: Math.asin(y / SKY_RADIUS) };
    }
  }

  zoom(factor) {
    if (this.mixTarget) this.view.dist = clamp(this.view.dist * factor, LIMITS.dist);
    else this.view.fov = clamp(this.view.fov * factor, LIMITS.fov);
  }

  resetView() {
    Object.assign(this.view, DEFAULT_VIEW);
    this.goto = null;
  }

  setInteractive(value) {
    this.interactive = value;
    this.canvas.style.pointerEvents = value ? 'auto' : 'none';
    if (!value) {
      this.drag = null;
      this.hoveredId = null;
    }
  }

  /** Pose inicial: vista lejana, sin estrellas del catálogo, fondo invisible y giro lento. */
  enterIntroPose() {
    this.tweens = [];
    this.mixTarget = 1;
    this.mix = 1;
    Object.assign(this.view, INTRO_VIEW);
    this.starAlpha = 0;
    this.backdropAlpha = 0;
    this.drift = !this.reducedMotion;
    this.goto = null;
    this.setInteractive(false);
  }

  /** Del negro a los puntos blancos del fondo. */
  revealBackdrop(ms = 2200) {
    this.tween(this, 'backdropAlpha', 1, ms);
  }

  /** El plano se acerca hasta la vista del mapa y aparecen las estrellas del catálogo. */
  explore(onArrive) {
    const ms = 2600;
    const v = DEFAULT_VIEW;
    this.drift = false;
    this.goto = null;
    this.setMode('sky', ms);
    this.tween(this.view, 'yaw', this.view.yaw + wrapAngle(v.yaw - this.view.yaw), ms);
    this.tween(this.view, 'pitch', v.pitch, ms);
    this.tween(this.view, 'fov', v.fov, ms);
    this.tween(this.view, 'dist', v.dist, ms);
    this.tween(this, 'starAlpha', 1, 1600, { delay: 1000, onDone: onArrive });
  }

  dispose() {
    cancelAnimationFrame(this.raf);
    this.abort.abort();
    this.resizeObserver.disconnect();
    this.scene.traverse((o) => {
      o.geometry?.dispose();
      o.material?.dispose();
    });
    this.renderer.dispose();
    this.renderer.forceContextLoss();
    this.labelLayer.replaceChildren();
  }

  // ---- Animaciones por tiempo ----
  tween(obj, key, to, ms, { delay = 0, onDone } = {}) {
    this.tweens = this.tweens.filter((t) => !(t.obj === obj && t.key === key));
    if (this.reducedMotion || ms <= 0) {
      obj[key] = to;
      onDone?.();
      return;
    }
    this.tweens.push({ obj, key, from: obj[key], to, t0: performance.now() + delay, ms, onDone });
  }

  runTweens(now) {
    const finished = [];
    for (const t of this.tweens) {
      if (now < t.t0) continue;
      const k = Math.min(1, (now - t.t0) / t.ms);
      t.obj[t.key] = t.from + (t.to - t.from) * easeInOut(k);
      if (k >= 1) finished.push(t);
    }
    if (finished.length) {
      this.tweens = this.tweens.filter((t) => !finished.includes(t));
      finished.forEach((t) => t.onDone?.());
    }
  }

  // ---- Construcción ----
  buildStars() {
    const n = this.stars.length;
    const sky = new Float32Array(n * 3);
    const space = new Float32Array(n * 3);
    const color = new Float32Array(n * 3);
    this.stars.forEach((s, i) => {
      sky.set(s.skyPos, i * 3);
      space.set(s.spacePos, i * 3);
      new THREE.Color(s.color).toArray(color, i * 3);
    });
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(sky, 3));
    geo.setAttribute('aSky', new THREE.BufferAttribute(sky, 3));
    geo.setAttribute('aSpace', new THREE.BufferAttribute(space, 3));
    geo.setAttribute('aColor', new THREE.BufferAttribute(color, 3));
    geo.setAttribute('aSize', new THREE.BufferAttribute(new Float32Array(n).fill(15), 1));
    this.uniforms = { uMix: { value: 0 }, uAlpha: { value: 1 }, uPx: { value: this.renderer.getPixelRatio() } };
    const points = new THREE.Points(
      geo,
      new THREE.ShaderMaterial({
        uniforms: this.uniforms,
        vertexShader: VERTEX,
        fragmentShader: FRAGMENT,
        transparent: true,
        depthWrite: false,
        blending: THREE.AdditiveBlending,
      }),
    );
    points.frustumCulled = false; // la posición real cambia en el shader
    this.scene.add(points);
  }

  buildBackdrop() {
    const count = 1800;
    const pos = new Float32Array(count * 3);
    for (let i = 0; i < count; i++) {
      const u = Math.random() * 2 - 1;
      const phi = Math.random() * Math.PI * 2;
      const r = Math.sqrt(1 - u * u);
      pos.set([r * Math.cos(phi) * 1200, u * 1200, r * Math.sin(phi) * 1200], i * 3);
    }
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    this.backdropMaterial = new THREE.PointsMaterial({ size: 1.2, sizeAttenuation: false, color: 0x8890aa, transparent: true });
    this.scene.add(new THREE.Points(geo, this.backdropMaterial));
  }

  buildRing(radius, color) {
    const pts = Array.from({ length: 128 }, (_, i) => {
      const a = (i / 128) * Math.PI * 2;
      return new THREE.Vector3(Math.cos(a) * radius, 0, Math.sin(a) * radius);
    });
    const line = new THREE.LineLoop(
      new THREE.BufferGeometry().setFromPoints(pts),
      new THREE.LineBasicMaterial({ color, transparent: true, opacity: 0 }),
    );
    this.scene.add(line);
    return line;
  }

  buildLabels() {
    this.labels = new Map(
      this.stars.map((s) => {
        const el = document.createElement('div');
        el.className = 'star-label';
        el.textContent = s.name;
        this.labelLayer.appendChild(el);
        return [s.id, el];
      }),
    );
    this.ring = document.createElement('div');
    this.ring.className = 'star-ring';
    this.labelLayer.appendChild(this.ring);
  }

  bindEvents() {
    this.abort = new AbortController();
    const opts = { signal: this.abort.signal };
    const c = this.canvas;
    c.addEventListener('pointerdown', (e) => {
      this.drag = { x: e.clientX, y: e.clientY, x0: e.clientX, y0: e.clientY };
      c.setPointerCapture(e.pointerId);
      this.goto = null;
    }, opts);
    c.addEventListener('pointermove', (e) => {
      if (this.drag) {
        this.view.yaw += (e.clientX - this.drag.x) * 0.005;
        this.view.pitch = Math.max(-1.5, Math.min(1.5, this.view.pitch + (e.clientY - this.drag.y) * 0.005));
        this.drag.x = e.clientX;
        this.drag.y = e.clientY;
      } else {
        this.hoveredId = this.nearest(e, PICK_RADIUS_PX)?.id ?? null;
        c.style.cursor = this.hoveredId ? 'pointer' : 'grab';
      }
    }, opts);
    c.addEventListener('pointerup', (e) => {
      const moved = this.drag && Math.hypot(e.clientX - this.drag.x0, e.clientY - this.drag.y0);
      this.drag = null;
      if (moved !== null && moved < 5) {
        const star = this.nearest(e, PICK_RADIUS_PX);
        if (star) this.onSelect(star.id);
      }
    }, opts);
    c.addEventListener('wheel', (e) => {
      e.preventDefault();
      this.zoom(Math.exp(e.deltaY * 0.001));
    }, { ...opts, passive: false });
    this.resizeObserver = new ResizeObserver(() => this.resize());
    this.resizeObserver.observe(c.parentElement);
  }

  resize() {
    const { clientWidth: w, clientHeight: h } = this.canvas.parentElement;
    this.renderer.setSize(w, h, false);
    this.camera.aspect = w / h;
    this.camera.updateProjectionMatrix();
  }

  // ---- Proyección y selección (O(n): para >10 000 estrellas pasar a índice espacial) ----
  project(star) {
    const p = this.tmpA.fromArray(star.skyPos).lerp(this.tmpB.fromArray(star.spacePos), this.mix).project(this.camera);
    return { x: ((p.x + 1) / 2) * this.canvas.clientWidth, y: ((1 - p.y) / 2) * this.canvas.clientHeight, visible: p.z < 1 };
  }

  nearest(event, radius) {
    const rect = this.canvas.getBoundingClientRect();
    const x = event.clientX - rect.left;
    const y = event.clientY - rect.top;
    let best = null;
    let bestDist = radius;
    for (const s of this.stars) {
      const q = this.project(s);
      const d = q.visible ? Math.hypot(q.x - x, q.y - y) : Infinity;
      if (d < bestDist) {
        bestDist = d;
        best = s;
      }
    }
    return best;
  }

  // ---- Bucle ----
  updateCamera() {
    const { yaw, pitch, fov, dist } = this.view;
    const f = new THREE.Vector3(Math.cos(pitch) * Math.cos(yaw), Math.sin(pitch), -Math.cos(pitch) * Math.sin(yaw));
    const pos = f.clone().multiplyScalar(-dist * this.mix);
    this.camera.position.copy(pos);
    this.camera.lookAt(pos.clone().add(f.multiplyScalar(SKY_RADIUS * (1 - this.mix) + dist * this.mix)));
    this.camera.fov = fov * (1 - this.mix) + 55 * this.mix;
    this.camera.updateProjectionMatrix();
  }

  updateLabels() {
    const { clientWidth: w, clientHeight: h } = this.canvas;
    for (const s of this.stars) {
      const el = this.labels.get(s.id);
      const q = this.project(s);
      const onScreen = q.visible && q.x > 0 && q.y > 0 && q.x < w && q.y < h;
      el.style.display = onScreen ? 'block' : 'none';
      if (!onScreen) continue;
      el.style.transform = `translate(${q.x + 14}px, ${q.y - 8}px)`;
      el.classList.toggle('is-selected', s.id === this.selectedId);
      el.textContent = s.id === this.hoveredId ? `${s.name} · ${s.spectralType.split('-')[0]} · ${Math.round(s.frequencyHz)} Hz` : s.name;
      if (s.id === this.selectedId) this.ring.style.transform = `translate(${q.x}px, ${q.y}px)`;
    }
    const sel = this.stars.find((s) => s.id === this.selectedId);
    this.ring.style.display = sel && this.project(sel).visible ? 'block' : 'none';
  }

  tick(now) {
    this.raf = requestAnimationFrame(this.tick);
    const dt = Math.min((now - this.lastT) / 1000, 0.05);
    this.lastT = now;
    this.runTweens(performance.now());
    if (this.drift && !this.drag) this.view.yaw += dt * 0.03;

    this.uniforms.uMix.value = this.mix;
    this.uniforms.uAlpha.value = this.starAlpha;
    this.backdropMaterial.opacity = this.backdropAlpha;

    if (this.goto) {
      const dy = wrapAngle(this.goto.yaw - this.view.yaw);
      this.view.yaw += dy * 0.1;
      this.view.pitch += (this.goto.pitch - this.view.pitch) * 0.1;
      if (Math.abs(dy) < 0.002) this.goto = null;
    }
    this.updateCamera();
    this.equator.material.opacity = 0.5 * (1 - this.mix) * this.starAlpha;
    this.rings.forEach((r) => (r.material.opacity = 0.7 * this.mix * this.starAlpha));
    this.renderer.render(this.scene, this.camera);
    this.updateLabels();
  }
}

/** Viaje por el espacio. Se puede saltar con el botón o con Esc. Al terminar avisa con onDone. */
export default function FlightIntro({ onDone }) {
  const canvasRef = useRef(null);
  const onDoneRef = useRef(onDone);
  onDoneRef.current = onDone;
  const [leaving, setLeaving] = useState(false);

  useEffect(() => {
    const scene = new FlightScene({ canvas: canvasRef.current, onEnd: () => setLeaving(true) });
    return () => scene.dispose();
  }, []);

  useEffect(() => {
    const onKey = (e) => e.key === 'Escape' && setLeaving(true);
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, []);

  // Fundido a negro antes de entregar el control (evita un corte brusco al saltar).
  useEffect(() => {
    if (!leaving) return undefined;
    const id = setTimeout(() => onDoneRef.current(), 600);
    return () => clearTimeout(id);
  }, [leaving]);

  return (
    <div className={`flight ${leaving ? 'is-leaving' : ''}`}>
      <canvas ref={canvasRef} className="flight__canvas" aria-hidden="true" />
      <button className="btn flight__skip" onClick={() => setLeaving(true)}>Saltar intro</button>
    </div>
  );
}

import { forwardRef, useEffect, useImperativeHandle, useRef } from 'react';
import { SkyScene } from '../three/SkyScene.js';

/**
 * Puente React <-> motor 3D. El motor se crea una vez (siempre en la pose de introducción);
 * las props solo disparan métodos. `interactive` habilita el mapa cuando termina la entrada.
 */
const StarCanvas = forwardRef(function StarCanvas({ stars, mode, selectedId, onSelect, interactive }, ref) {
  const canvasRef = useRef(null);
  const labelsRef = useRef(null);
  const sceneRef = useRef(null);
  const onSelectRef = useRef(onSelect);
  onSelectRef.current = onSelect;

  useImperativeHandle(ref, () => ({
    revealBackdrop: (ms) => sceneRef.current?.revealBackdrop(ms),
    explore: (onArrive) => sceneRef.current?.explore(onArrive),
  }), []);

  useEffect(() => {
    const scene = new SkyScene({
      canvas: canvasRef.current,
      labelLayer: labelsRef.current,
      stars,
      onSelect: (id) => onSelectRef.current(id),
      reducedMotion: window.matchMedia('(prefers-reduced-motion: reduce)').matches,
    });
    scene.enterIntroPose();
    sceneRef.current = scene;
    return () => scene.dispose();
  }, [stars]);

  // El modo solo se aplica cuando el mapa es interactivo: durante la intro manda la coreografía.
  useEffect(() => {
    if (interactive) sceneRef.current?.setMode(mode);
  }, [mode, interactive]);
  useEffect(() => {
    sceneRef.current?.setInteractive(interactive);
  }, [interactive]);
  useEffect(() => {
    sceneRef.current?.setSelected(selectedId);
  }, [selectedId]);

  return (
    <div className="map">
      <canvas ref={canvasRef} className="map__canvas" aria-label="Mapa estelar interactivo" />
      <div ref={labelsRef} className="map__labels" aria-hidden="true" />
      <div className="zoom">
        <button className="btn btn--icon" aria-label="Acercar" onClick={() => sceneRef.current?.zoom(0.85)}>+</button>
        <button className="btn btn--icon" aria-label="Alejar" onClick={() => sceneRef.current?.zoom(1.18)}>−</button>
        <button className="btn btn--icon" aria-label="Restablecer vista" onClick={() => sceneRef.current?.resetView()}>⌂</button>
      </div>
      <p className="hint">Arrastra para orbitar · rueda para zoom · clic en una estrella para escucharla</p>
    </div>
  );
});

export default StarCanvas;

import { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import catalog from './data/stars.json';
import { enrichStar } from './domain/starMath.js';
import { useStarPlayer } from './hooks/useStarPlayer.js';
import FlightIntro from './components/FlightIntro.jsx';
import Welcome from './components/Welcome.jsx';
import Header from './components/Header.jsx';
import StarCanvas from './components/StarCanvas.jsx';
import StarCatalog from './components/StarCatalog.jsx';
import StarDetail from './components/StarDetail.jsx';
import PlayerBar from './components/PlayerBar.jsx';

// Fases: flight (video) -> reveal (aparecen las estrellas) -> welcome (logo + Explorar)
//        -> entering (viaje al mapa) -> ready (interfaz completa)
const SEEN_KEY = 'astrowave:intro-seen';
const prefersReducedMotion = () => window.matchMedia('(prefers-reduced-motion: reduce)').matches;

function introAlreadySeen() {
  if (new URLSearchParams(window.location.search).has('intro')) return false; // ?intro fuerza el vuelo
  try {
    return localStorage.getItem(SEEN_KEY) === '1';
  } catch {
    return false;
  }
}

export default function App() {
  const stars = useMemo(() => catalog.map(enrichStar), []);
  const [phase, setPhase] = useState(() => (prefersReducedMotion() || introAlreadySeen() ? 'reveal' : 'flight'));
  const [mode, setMode] = useState('sky');
  const [selectedId, setSelectedId] = useState(stars[0].id);
  const mapRef = useRef(null);
  const player = useStarPlayer();
  const { play, stop, unlock } = player;

  const selected = stars.find((s) => s.id === selectedId);

  const endFlight = useCallback(() => {
    try {
      localStorage.setItem(SEEN_KEY, '1');
    } catch {
      /* sin almacenamiento: la intro se repetirá, no es grave */
    }
    setPhase('reveal');
  }, []);

  // Revelación: el fondo sale del negro y poco después entra la bienvenida (se solapan a propósito).
  useEffect(() => {
    if (phase !== 'reveal') return undefined;
    mapRef.current?.revealBackdrop();
    const id = setTimeout(() => setPhase('welcome'), prefersReducedMotion() ? 0 : 1400);
    return () => clearTimeout(id);
  }, [phase]);

  const explore = () => {
    if (phase !== 'welcome') return;
    unlock().catch(() => {}); // el clic del usuario habilita el audio
    setPhase('entering');
    mapRef.current?.explore(() => setPhase('ready'));
  };

  const select = useCallback(
    (id) => {
      setSelectedId(id);
      play(stars.find((s) => s.id === id));
    },
    [play, stars],
  );

  const step = (delta) => {
    const i = stars.findIndex((s) => s.id === selectedId);
    select(stars[(i + delta + stars.length) % stars.length].id);
  };

  return (
    <div className="app" data-phase={phase}>
      {phase === 'flight' && <FlightIntro onDone={endFlight} />}
      <Header mode={mode} onModeChange={setMode} muted={player.muted} onToggleMuted={player.toggleMuted} />
      <main className="stage">
        <StarCanvas ref={mapRef} stars={stars} mode={mode} selectedId={selectedId} onSelect={select} interactive={phase === 'ready'} />
        <StarCatalog stars={stars} selectedId={selectedId} onSelect={select} />
        <StarDetail star={selected} />
      </main>
      <PlayerBar
        star={selected}
        playing={player.playingId === selectedId}
        engine={player.engine}
        volume={player.volume}
        onVolume={player.setVolume}
        onToggle={() => (player.playingId ? stop() : play(selected))}
        onPrev={() => step(-1)}
        onNext={() => step(1)}
      />
      {(phase === 'welcome' || phase === 'entering') && <Welcome leaving={phase === 'entering'} onExplore={explore} />}
    </div>
  );
}
